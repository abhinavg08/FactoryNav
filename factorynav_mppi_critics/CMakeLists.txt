cmake_minimum_required(VERSION 3.8)
project(factorynav_mppi_critics)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(pluginlib REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
add_library(factorynav_dynamic_critic SHARED src/dynamic_obstacle_critic.cpp)
target_include_directories(factorynav_dynamic_critic PRIVATE include)
ament_target_dependencies(factorynav_dynamic_critic rclcpp rclcpp_lifecycle pluginlib sensor_msgs nav2_mppi_controller)
pluginlib_export_plugin_description_file(nav2_mppi_controller critics.xml)
install(TARGETS factorynav_dynamic_critic LIBRARY DESTINATION lib)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_temporal_cost test/test_temporal_cost.cpp)
  target_include_directories(test_temporal_cost PRIVATE include)
  add_test(NAME temporal_collision_timing COMMAND test_temporal_cost)
endif()
ament_package()
